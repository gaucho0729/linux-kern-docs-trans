ブロックIOの優先度¶

はじめに¶

cfq v3（cfq-tsまたはタイムスライスされたcfq）の導入により、ファイルの読み取りに基本的なI/O優先度がサポートされています。これにより、ユーザーはCPUスケジューリングのように長年可能だったように、プロセスまたはプロセスグループのI/O優先度を設定できます。このドキュメントは主にcfqの現在の機能を説明しており、他のI/OスケジューラはまだI/O優先度をサポートしていません。

スケジューリングクラス¶

CFQはプロセスのI/Oをどのように処理するかを決定する3つの汎用的なスケジューリングクラスを実装しています。

IOPRIO_CLASS_RT: これはリアルタイムI/Oクラスです。このスケジューリングクラスはシステム内の他のクラスよりも高い優先度が与えられ、このクラスのプロセスは常にディスクへの最初のアクセスを獲得します。そのため、注意深く使用する必要があります。1つのI/Oリアルタイムプロセスがシステム全体を食い尽くす可能性があります。RTクラス内では、プロセスがディスクにどのくらいの時間アクセスするかを決定する8段階のクラスデータがあります。今後はパフォーマンスに直接マッピングされるよう、望ましいデータレートを指定する形に変更される可能性があります。

IOPRIO_CLASS_BE: これはベストエフォートスケジューリングクラスで、特定のI/O優先度を設定していないプロセスのデフォルトです。クラスデータはプロセスが取得するI/O帯域幅を決定し、CPUのniceレベルとほぼ同様に直接マッピングされていますが、より粗く実装されています。0は最高のBE優先度レベルで、7は最低です。CPU niceレベルとI/O niceレベルのマッピングは、io_nice = (cpu_nice + 20) / 5 で決定されます。

IOPRIO_CLASS_IDLE: これはアイドルスケジューリングクラスで、このレベルで実行されているプロセスは、他のプロセスがディスクを使用していないときにのみI/O時間を受けます。アイドルクラスにはクラスデータが存在せず、ここでは適用されません。

ツール¶

以下にioniceツールのサンプルを示します。使用法:

# ionice -c<class> -n<level> -p<pid>

pidが指定されていない場合、現在のプロセスが仮定されます。I/O優先度の設定はfork時に継承されるため、ioniceを使用してプロセスを特定のレベルで起動できます:

# ionice -c2 -n0 /bin/ls

lsをベストエフォートスケジューリングクラスで最高優先度で実行します。すでに実行中のプロセスの場合、pidを指定できます:

# ionice -c1 -n2 -p100

pid 100をリアルタイムスケジューリングクラスで優先度2で実行します。

ionice.cツール:

#include <stdio.h>
#include <stdlib.h>
#include <errno.h>
#include <getopt.h>
#include <unistd.h>
#include <sys/ptrace.h>
#include <asm/unistd.h>

extern int sys_ioprio_set(int, int, int);
extern int sys_ioprio_get(int, int);

#if defined(__i386__)
#define __NR_ioprio_set               289
#define __NR_ioprio_get               290
#elif defined(__ppc__)
#define __NR_ioprio_set               273
#define __NR_ioprio_get               274
#elif defined(__x86_64__)
#define __NR_ioprio_set               251
#define __NR_ioprio_get               252
#elif defined(__ia64__)
#define __NR_ioprio_set               1274
#define __NR_ioprio_get               1275
#else
#error "Unsupported arch"
#endif

static inline int ioprio_set(int which, int who, int ioprio)
{
      return syscall(__NR_ioprio_set, which, who, ioprio);
}

static inline int ioprio_get(int which, int who)
{
      return syscall(__NR_ioprio_get, which, who);
}

enum {
      IOPRIO_CLASS_NONE,
      IOPRIO_CLASS_RT,
      IOPRIO_CLASS_BE,
      IOPRIO_CLASS_IDLE,
};

enum {
      IOPRIO_WHO_PROCESS = 1,
      IOPRIO_WHO_PGRP,
      IOPRIO_WHO_USER,
};

#define IOPRIO_CLASS_SHIFT    13

const char *to_prio[] = { "none", "realtime", "best-effort", "idle", };

int main(int argc, char *argv[])
{
      int ioprio = 4, set = 0, ioprio_class = IOPRIO_CLASS_BE;
      int c, pid = 0;

      while ((c = getopt(argc, argv, "+n:c:p:")) != EOF) {
              switch (c) {
              case 'n':
                      ioprio = strtol(optarg, NULL, 10);
                      set = 1;
                      break;
              case 'c':
                      ioprio_class = strtol(optarg, NULL, 10);
                      set = 1;
                      break;
              case 'p':
                      pid = strtol(optarg, NULL, 10);
                      break;
              }
      }

      switch (ioprio_class) {
              case IOPRIO_CLASS_NONE:
                      ioprio_class = IOPRIO_CLASS_BE;
                      break;
              case IOPRIO_CLASS_RT:
              case IOPRIO_CLASS_BE:
                      break;
              case IOPRIO_CLASS_IDLE:
                      ioprio = 7;
                      break;
              default:
                      printf("bad prio class %d\n", ioprio_class);
                      return 1;
      }

      if (!set) {
              if (!pid && argv[optind])
                      pid = strtol(argv[optind], NULL, 10);

              ioprio = ioprio_get(IOPRIO_WHO_PROCESS, pid);

              printf("pid=%d, %d\n", pid, ioprio);

              if (ioprio == -1)
                      perror("ioprio_get");
              else {
                      ioprio_class = ioprio >> IOPRIO_CLASS_SHIFT;
                      ioprio = ioprio & 0xff;
                      printf("%s: prio %d\n", to_prio[ioprio_class], ioprio);
              }
      } else {
              if (ioprio_set(IOPRIO_WHO_PROCESS, pid, ioprio | ioprio_class << IOPRIO_CLASS_SHIFT) == -1) {
                      perror("ioprio_set");
                      return 1;
              }

              if (argv[optind])
                      execvp(argv[optind], &argv[optind]);
      }

      return 0;
}
