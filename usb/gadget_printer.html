Linux USBプリンタギアドライバ¶

06/04/2007

Copyright (C) 2007 Craig W. Nadler <craig@nadler.us>

一般¶

このドライバーは、Linuxを埋め込みOSとして使用してプリンターファームウェアを開発する場合に使用できます。このドライバーは、Linuxホストシステムでプリンターを使用するためのものではありません。

USBデバイスコントローラーと、そのデバイスコントローラー用のLinuxドライバーが必要です。このドライバーは、Linux USB Gadget APIを使用した「デバイスクラス」ドライバーを受容する必要があります。USBデバイスコントローラーのドライバーがロードされた後、プリンターガジェットドライバーをロードします。これにより、USBホストにプリンターアクセスインターフェースが提供されます。

このドライバーは、ユーザーモードで動作するプリンターファームウェア用に構築されています。ユーザーモードのプリンターファームウェアは、デバイスファイルを介してカーネルモードのプリンターガジェットドライバーからデータを読み書きします。USBホストがプリンターステータスを取得するデバイスリクエストを送信すると、プリンターがプリンターステータスバイトを返します。ユーザー空間のファームウェアは、/dev/g_printer デバイスファイルを介してこのステータスバイトを読み書きできます。ブロッキングおよびノンブロッキングの読み書きコールがサポートされています。

このドライバーの使用方法¶

USBデバイスコントローラードライバーとプリンターガジェットドライバーをロードする方法です。以下の例はNetchip 2280 USBデバイスコントローラードライバーを使用しています:

modprobe net2280
modprobe g_printer

プリンターガジェットをロードする際には、次のコマンドラインパラメータを使用できます（例: modprobe g_printer idVendor=0x0525 idProduct=0xa4a8）:

idVendor

これはデバイス記述子で使用されるベンダーアイディです。デフォルトはNetchipのベンダーアイディ0x0525です。製品をリリースする前に、自分のベンダーアイディに変更してください。製品をリリースする予定がある場合は、ベンダーアイディをまだ持っていない場合は、www.usb.org で詳細をご確認ください。

idProduct

これはデバイス記述子で使用される製品アイディです。デフォルトは0xa4a8です。他のUSB製品と重複しないIDに変更することをお勧めします。製品の番号は0x0001から始めるのが良いでしょう。

bcdDevice

これは製品のバージョン番号です。ファームウェアのバージョンをここに記載するのが良いでしょう。

iManufacturer

ベンダーの名前を含む文字列です。

iProduct

製品名を含む文字列です。

iSerialNum

シリアル番号を含む文字列です。このシリアル番号は製品の各ユニットごとに変更してください。

iPNPstring

このプリンター用のPNP ID文字列です。このプリンター製品のPNP ID文字列をコマンドラインで設定するか、ハードコードしてください。

qlen

各エンドポイントで使用する8kバッファの数。デフォルト値は10ですが、製品に合わせて調整してください。また、各バッファのサイズも製品に合わせて調整することを検討してください。

サンプルコードの使用¶

このサンプルコードは、プリンタエンジンではなく標準出力に話します。

以下のテストコードをコンパイルするには:

  1. これをprn_example.cというファイルに保存してください。

  2. 以下のコマンドでコードをコンパイルしてください:

    gcc prn_example.c -o prn_example

ホストから標準出力にプリンタデータを読み込むには:

# prn_example -read_data

ファイル（data_file）からホストにプリンタデータを書き込むには:

# cat data_file | prn_example -write_data

ギャクドライバーの現在のプリンタステータスを取得するには::

# prn_example -get_status

Printer status is:
     Printer is NOT Selected
     Paper is Out
     Printer OK

プリンタを選択済み/オンラインに設定するには:

# prn_example -selected

プリンタを未選択/オフラインに設定するには:

# prn_example -not_selected

紙が切れている状態に紙状態を設定するには:

# prn_example -paper_out

紙が挿入されている状態に紙状態を設定するには:

# prn_example -paper_loaded

エラーステータスをプリンタOKに設定するには:

# prn_example -no_error

エラーステータスをエラーに設定するには:

# prn_example -error

例コード¶

#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <linux/poll.h>
#include <sys/ioctl.h>
#include <linux/usb/g_printer.h>

#define PRINTER_FILE                  "/dev/g_printer"
#define BUF_SIZE                      512


/*
 * 'usage()' - Show program usage.
 */

static void
usage(const char *option)             /* I - Option string or NULL */
{
      if (option) {
              fprintf(stderr,"prn_example: Unknown option \"%s\"!\n",
                              option);
      }

      fputs("\n", stderr);
      fputs("Usage: prn_example -[options]\n", stderr);
      fputs("Options:\n", stderr);
      fputs("\n", stderr);
      fputs("-get_status    Get the current printer status.\n", stderr);
      fputs("-selected      Set the selected status to selected.\n", stderr);
      fputs("-not_selected  Set the selected status to NOT selected.\n",
                      stderr);
      fputs("-error         Set the error status to error.\n", stderr);
      fputs("-no_error      Set the error status to NO error.\n", stderr);
      fputs("-paper_out     Set the paper status to paper out.\n", stderr);
      fputs("-paper_loaded  Set the paper status to paper loaded.\n",
                      stderr);
      fputs("-read_data     Read printer data from driver.\n", stderr);
      fputs("-write_data    Write printer sata to driver.\n", stderr);
      fputs("-NB_read_data  (Non-Blocking) Read printer data from driver.\n",
                      stderr);
      fputs("\n\n", stderr);

      exit(1);
}


static int
read_printer_data()
{
      struct pollfd   fd[1];

      /* Open device file for printer gadget. */
      fd[0].fd = open(PRINTER_FILE, O_RDWR);
      if (fd[0].fd < 0) {
              printf("Error %d opening %s\n", fd[0].fd, PRINTER_FILE);
              close(fd[0].fd);
              return(-1);
      }

      fd[0].events = POLLIN | POLLRDNORM;

      while (1) {
              static char buf[BUF_SIZE];
              int bytes_read;
              int retval;

              /* Wait for up to 1 second for data. */
              retval = poll(fd, 1, 1000);

              if (retval && (fd[0].revents & POLLRDNORM)) {

                      /* Read data from printer gadget driver. */
                      bytes_read = read(fd[0].fd, buf, BUF_SIZE);

                      if (bytes_read < 0) {
                              printf("Error %d reading from %s\n",
                                              fd[0].fd, PRINTER_FILE);
                              close(fd[0].fd);
                              return(-1);
                      } else if (bytes_read > 0) {
                              /* Write data to standard OUTPUT (stdout). */
                              fwrite(buf, 1, bytes_read, stdout);
                              fflush(stdout);
                      }

              }

      }

      /* Close the device file. */
      close(fd[0].fd);

      return 0;
}


static int
write_printer_data()
{
      struct pollfd   fd[1];

      /* Open device file for printer gadget. */
      fd[0].fd = open (PRINTER_FILE, O_RDWR);
      if (fd[0].fd < 0) {
              printf("Error %d opening %s\n", fd[0].fd, PRINTER_FILE);
              close(fd[0].fd);
              return(-1);
      }

      fd[0].events = POLLOUT | POLLWRNORM;

      while (1) {
              int retval;
              static char buf[BUF_SIZE];
              /* Read data from standard INPUT (stdin). */
              int bytes_read = fread(buf, 1, BUF_SIZE, stdin);

              if (!bytes_read) {
                      break;
              }

              while (bytes_read) {

                      /* Wait for up to 1 second to sent data. */
                      retval = poll(fd, 1, 1000);

                      /* Write data to printer gadget driver. */
                      if (retval && (fd[0].revents & POLLWRNORM)) {
                              retval = write(fd[0].fd, buf, bytes_read);
                              if (retval < 0) {
                                      printf("Error %d writing to %s\n",
                                                      fd[0].fd,
                                                      PRINTER_FILE);
                                      close(fd[0].fd);
                                      return(-1);
                              } else {
                                      bytes_read -= retval;
                              }

                      }

              }

      }

      /* Wait until the data has been sent. */
      fsync(fd[0].fd);

      /* Close the device file. */
      close(fd[0].fd);

      return 0;
}


static int
read_NB_printer_data()
{
      int             fd;
      static char     buf[BUF_SIZE];
      int             bytes_read;

      /* Open device file for printer gadget. */
      fd = open(PRINTER_FILE, O_RDWR|O_NONBLOCK);
      if (fd < 0) {
              printf("Error %d opening %s\n", fd, PRINTER_FILE);
              close(fd);
              return(-1);
      }

      while (1) {
              /* Read data from printer gadget driver. */
              bytes_read = read(fd, buf, BUF_SIZE);
              if (bytes_read <= 0) {
                      break;
              }

              /* Write data to standard OUTPUT (stdout). */
              fwrite(buf, 1, bytes_read, stdout);
              fflush(stdout);
      }

      /* Close the device file. */
      close(fd);

      return 0;
}


static int
get_printer_status()
{
      int     retval;
      int     fd;

      /* Open device file for printer gadget. */
      fd = open(PRINTER_FILE, O_RDWR);
      if (fd < 0) {
              printf("Error %d opening %s\n", fd, PRINTER_FILE);
              close(fd);
              return(-1);
      }

      /* Make the IOCTL call. */
      retval = ioctl(fd, GADGET_GET_PRINTER_STATUS);
      if (retval < 0) {
              fprintf(stderr, "ERROR: Failed to set printer status\n");
              return(-1);
      }

      /* Close the device file. */
      close(fd);

      return(retval);
}


static int
set_printer_status(unsigned char buf, int clear_printer_status_bit)
{
      int     retval;
      int     fd;

      retval = get_printer_status();
      if (retval < 0) {
              fprintf(stderr, "ERROR: Failed to get printer status\n");
              return(-1);
      }

      /* Open device file for printer gadget. */
      fd = open(PRINTER_FILE, O_RDWR);

      if (fd < 0) {
              printf("Error %d opening %s\n", fd, PRINTER_FILE);
              close(fd);
              return(-1);
      }

      if (clear_printer_status_bit) {
              retval &= ~buf;
      } else {
              retval |= buf;
      }

      /* Make the IOCTL call. */
      if (ioctl(fd, GADGET_SET_PRINTER_STATUS, (unsigned char)retval)) {
              fprintf(stderr, "ERROR: Failed to set printer status\n");
              return(-1);
      }

      /* Close the device file. */
      close(fd);

      return 0;
}


static int
display_printer_status()
{
      char    printer_status;

      printer_status = get_printer_status();
      if (printer_status < 0) {
              fprintf(stderr, "ERROR: Failed to get printer status\n");
              return(-1);
      }

      printf("Printer status is:\n");
      if (printer_status & PRINTER_SELECTED) {
              printf("     Printer is Selected\n");
      } else {
              printf("     Printer is NOT Selected\n");
      }
      if (printer_status & PRINTER_PAPER_EMPTY) {
              printf("     Paper is Out\n");
      } else {
              printf("     Paper is Loaded\n");
      }
      if (printer_status & PRINTER_NOT_ERROR) {
              printf("     Printer OK\n");
      } else {
              printf("     Printer ERROR\n");
      }

      return(0);
}


int
main(int  argc, char *argv[])
{
      int     i;              /* Looping var */
      int     retval = 0;

      /* No Args */
      if (argc == 1) {
              usage(0);
              exit(0);
      }

      for (i = 1; i < argc && !retval; i ++) {

              if (argv[i][0] != '-') {
                      continue;
              }

              if (!strcmp(argv[i], "-get_status")) {
                      if (display_printer_status()) {
                              retval = 1;
                      }

              } else if (!strcmp(argv[i], "-paper_loaded")) {
                      if (set_printer_status(PRINTER_PAPER_EMPTY, 1)) {
                              retval = 1;
                      }

              } else if (!strcmp(argv[i], "-paper_out")) {
                      if (set_printer_status(PRINTER_PAPER_EMPTY, 0)) {
                              retval = 1;
                      }

              } else if (!strcmp(argv[i], "-selected")) {
                      if (set_printer_status(PRINTER_SELECTED, 0)) {
                              retval = 1;
                      }

              } else if (!strcmp(argv[i], "-not_selected")) {
                      if (set_printer_status(PRINTER_SELECTED, 1)) {
                              retval = 1;
                      }

              } else if (!strcmp(argv[i], "-error")) {
                      if (set_printer_status(PRINTER_NOT_ERROR, 1)) {
                              retval = 1;
                      }

              } else if (!strcmp(argv[i], "-no_error")) {
                      if (set_printer_status(PRINTER_NOT_ERROR, 0)) {
                              retval = 1;
                      }

              } else if (!strcmp(argv[i], "-read_data")) {
                      if (read_printer_data()) {
                              retval = 1;
                      }

              } else if (!strcmp(argv[i], "-write_data")) {
                      if (write_printer_data()) {
                              retval = 1;
                      }

              } else if (!strcmp(argv[i], "-NB_read_data")) {
                      if (read_NB_printer_data()) {
                              retval = 1;
                      }

              } else {
                      usage(argv[i]);
                      retval = 1;
              }
      }

      exit(retval);
}
