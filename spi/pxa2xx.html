PXA2xx SSPドライバでのSPI HOWTO¶

これはpxa2xx_spiドライバに関するミニガイドです。このドライバはPXA2xx同期シリアルポートをSPIマスターコントローラーに変換します（参照 LinuxカーネルSPIサポートの概要）。ドライバは以下の機能を備えています

  • 任意のPXA2xx SSPのサポート

  • SSP PIOおよびSSP DMAデータ転送

  • 外部および内部（SSPFRM）チップセレクト

  • 各スレーブデバイス（チップ）の設定

  • フルスリープ、フリーズ、再開のサポート

このドライバは「spi_message」 fifoをワークキューおよびタスクレットでサービスします。ワークキュー「pump_messages」はメッセージfifoを駆動し、タスクレット（pump_transfer）はSPIトランザクションをキューイングし、DMA/割り込み駆動の転送を設定・実行します

PXA2xxマスターコントローラーの宣言¶

通常、SPIマスターはarch/…/mach-/board-.cで「platform device」として定義されます。マスター設定はinclude/linux/spi/pxa2xx_spi.hに記載されているテーブルを介してドライバに渡されます：

struct pxa2xx_spi_controller {
      u16 num_chipselect;
      u8 enable_dma;
};

「pxa2xx_spi_controller.num_chipselect」フィールドは、このSPIマスターに接続されているスレーブデバイス（チップ）の数を決定します

「pxa2xx_spi_controller.enable_dma」フィールドはドライバにSSP DMAを使用する旨を通知します。これによりドライバはrx_channelおよびtx_channelの2つのDMAチャネルを取得します。rx_channelはtx_channelよりもDMAサービス優先度が高いです。詳しくは「PXA2xx Developer Manual」の「DMA Controller」セクションを参照してください

NSSPマスターサンプル¶

以下はPXA255 NSSPを使用したサンプル構成です：

static struct resource pxa_spi_nssp_resources[] = {
      [0] = {
              .start  = __PREG(SSCR0_P(2)), /* Start address of NSSP */
              .end    = __PREG(SSCR0_P(2)) + 0x2c, /* Range of registers */
              .flags  = IORESOURCE_MEM,
      },
      [1] = {
              .start  = IRQ_NSSP, /* NSSP IRQ */
              .end    = IRQ_NSSP,
              .flags  = IORESOURCE_IRQ,
      },
};

static struct pxa2xx_spi_controller pxa_nssp_master_info = {
      .num_chipselect = 1, /* Matches the number of chips attached to NSSP */
      .enable_dma = 1, /* Enables NSSP DMA */
};

static struct platform_device pxa_spi_nssp = {
      .name = "pxa2xx-spi", /* MUST BE THIS VALUE, so device match driver */
      .id = 2, /* Bus number, MUST MATCH SSP number 1..n */
      .resource = pxa_spi_nssp_resources,
      .num_resources = ARRAY_SIZE(pxa_spi_nssp_resources),
      .dev = {
              .platform_data = &pxa_nssp_master_info, /* Passed to driver */
      },
};

static struct platform_device *devices[] __initdata = {
      &pxa_spi_nssp,
};

static void __init board_init(void)
{
      (void)platform_add_device(devices, ARRAY_SIZE(devices));
}

スレーブデバイスの宣言¶

通常、各SPIスレーブ（チップ）はarch/…/mach-/board-.cで「spi_board_info」構造体（linux/spi/spi.hに記載）を使用して定義されます。詳しくは「LinuxカーネルSPIサポートの概要」をご参照ください

PXAに接続された各スレーブデバイスは、include/linux/spi/pxa2xx_spi.hに記載されている「pxa2xx_spi_chip」構造体を介してスレーブ固有の設定情報を提供します。pxa2xx_spiマスターコントローラードライバは、ドライバがスレーブデバイスと通信する際、この設定情報を使用します。すべてのフィールドはオプションです

struct pxa2xx_spi_chip {
      u8 tx_threshold;
      u8 rx_threshold;
      u8 dma_burst_size;
      u32 timeout;
      u8 enable_loopback;
      void (*cs_control)(u32 command);
};

「pxa2xx_spi_chip.tx_threshold」および「pxa2xx_spi_chip.rx_threshold」フィールドはSSPハードウェアFIFOを設定するために使用されます。これらのフィールドはpxa2xx_spiドライバの性能に極めて重要であり、誤設定はrx FIFOのオーバーフロー（特にPIOモード転送では顕著）を引き起こします。適切なデフォルト値は：

.tx_threshold = 8,
.rx_threshold = 8,

範囲は1〜16で、ゼロは「デフォルトを使用」を意味します

「pxa2xx_spi_chip.dma_burst_size」フィールドは、PXA2xx DMAエンジンを構成するために使用され、「spi_device.bits_per_word」フィールドに関連しています。PXA2xxの「開発者マニュアル」のDMAコントローラーおよびSSPコントローラーに関するセクションを参照し、正しい値を決定してください。バイト幅で転送が設定されたSSPは、値8を使用します。dma_burst_sizeが0の場合、ドライバは適切なデフォルト値を自動的に設定します。

「pxa2xx_spi_chip.timeout」フィールドは、SSP受信FIFO内の末尾バイトを効率的に処理するために使用されます。このフィールドの正しい値はSPIバス速度（「spi_board_info.max_speed_hz」）と特定のスレーブデバイスに依存します。注意してください。PXA2xx SSP 1は末尾バイトタイムアウトをサポートしておらず、末尾バイトについてはbusy-wait処理を実行しなければなりません。

「pxa2xx_spi_chip.enable_loopback」フィールドは、SSPポートを内部ループバックモードに設定するために使用されます。このモードでは、SSPコントローラーが内部的にSSPTXピンをSSPRXピンに接続します。これは初期設定テストに役立ちます。

「pxa2xx_spi_chip.cs_control」フィールドは、スレーブデバイスのチップセレクトをアサート/デアサートするためのボード固有の関数を指します。このフィールドがNULLの場合、pxa2xx_spiマスターコントローラードライバはSSPポートがSSPFRMを使用していると仮定します。

注意：SSPFRMを使用する場合、SPIドライバはチップセレクトを制御できません。そのため、各spi_transfer後にチップセレクトが解除されます。ほとんどのデバイスでは、メッセージ全体にわたってチップセレクトがアサートされている必要があります。これらのデバイスには、cs_controlを通じてSSPFRMをGPIOとして使用してください。

NSSP スレーブ サンプル¶

pxa2xx_spi_chip構造体は、pxa2xx_spiドライバに「spi_board_info.controller_data」フィールドを通じて渡されます。以下はPXA255 NSSPを使用したサンプル構成です。

/* Chip Select control for the CS8415A SPI slave device */
static void cs8415a_cs_control(u32 command)
{
      if (command & PXA2XX_CS_ASSERT)
              GPCR(2) = GPIO_bit(2);
      else
              GPSR(2) = GPIO_bit(2);
}

/* Chip Select control for the CS8405A SPI slave device */
static void cs8405a_cs_control(u32 command)
{
      if (command & PXA2XX_CS_ASSERT)
              GPCR(3) = GPIO_bit(3);
      else
              GPSR(3) = GPIO_bit(3);
}

static struct pxa2xx_spi_chip cs8415a_chip_info = {
      .tx_threshold = 8, /* SSP hardward FIFO threshold */
      .rx_threshold = 8, /* SSP hardward FIFO threshold */
      .dma_burst_size = 8, /* Byte wide transfers used so 8 byte bursts */
      .timeout = 235, /* See Intel documentation */
      .cs_control = cs8415a_cs_control, /* Use external chip select */
};

static struct pxa2xx_spi_chip cs8405a_chip_info = {
      .tx_threshold = 8, /* SSP hardward FIFO threshold */
      .rx_threshold = 8, /* SSP hardward FIFO threshold */
      .dma_burst_size = 8, /* Byte wide transfers used so 8 byte bursts */
      .timeout = 235, /* See Intel documentation */
      .cs_control = cs8405a_cs_control, /* Use external chip select */
};

static struct spi_board_info streetracer_spi_board_info[] __initdata = {
      {
              .modalias = "cs8415a", /* Name of spi_driver for this device */
              .max_speed_hz = 3686400, /* Run SSP as fast a possbile */
              .bus_num = 2, /* Framework bus number */
              .chip_select = 0, /* Framework chip select */
              .platform_data = NULL; /* No spi_driver specific config */
              .controller_data = &cs8415a_chip_info, /* Master chip config */
              .irq = STREETRACER_APCI_IRQ, /* Slave device interrupt */
      },
      {
              .modalias = "cs8405a", /* Name of spi_driver for this device */
              .max_speed_hz = 3686400, /* Run SSP as fast a possbile */
              .bus_num = 2, /* Framework bus number */
              .chip_select = 1, /* Framework chip select */
              .controller_data = &cs8405a_chip_info, /* Master chip config */
              .irq = STREETRACER_APCI_IRQ, /* Slave device interrupt */

static void __init streetracer_init(void)
{

DMA および PIO I/O のサポート¶

pxa2xx_spiドライバはDMAおよび割込み駆動型PIOメッセージ転送の両方をサポートします。ドライバはデフォルトでPIOモードを使用し、DMA転送は「pxa2xx_spi_controller」構造体の「enable_dma」フラグを設定することで有効になります。DMAモードは、コヒーレント型およびストリームベースのDMAマッピングをサポートします。

各「spi_transfer」ごとに使用するI/Oタイプを決定する論理は以下の通りです：

if !enable_dma then
      always use PIO transfers

if spi_message.len > 8191 then
      print "rate limited" warning
      use PIO transfers

if spi_message.is_dma_mapped and rx_dma_buf != 0 and tx_dma_buf != 0 then
      use coherent DMA mode

if rx_buf and tx_buf are aligned on 8 byte boundary then
      use streaming DMA mode

otherwise
      use PIO transfer
