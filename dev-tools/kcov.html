kcov: code coverage for fuzzing¶

kcov は、カバレッジをガイドするファジング（ランダムテスト）に適した形で、カーネルのコードカバレッジ情報を公開します。実行中のカーネルのカバレッジデータは「kcov」debugfs ファイルを通じてエクスポートされます。カバレッジの収集はタスク単位で有効化され、これにより単一のシステムコールの正確なカバレッジをキャプチャできます。

kcov は、可能な限り多くのカバレッジを収集することを目的としていません。代わりに、システムコール入力に依存した比較的安定したカバレッジを収集することを目指しています。この目標を達成するため、ソフト/ハード割込みではカバレッジを収集せず、カーネルの非決定性の部分（例: スケジューラ、ロック）のインストゥルメンテーションは無効にされています。

kcov は、インストルメンテーションされたコードからの比較演算子の収集も行えます（この機能は現在、カーネルが clang でコンパイルされている必要があります）。

前提条件¶

全体のカーネルのカバレッジデータを取得するには：

CONFIG_KCOV=y

CONFIG_KCOV は gcc 6.1.0 またはそれ以降が必要です。

比較演算子を収集する場合は、以下を設定してください：

CONFIG_KCOV_ENABLE_COMPARISONS=y

特定のファイルまたはディレクトリのプロファイリングを有効にするには、対応するカーネルMakefileに以下の行を追加してください：

mount -t debugfs none /sys/kernel/debug

カバレッジの収集¶

以下のプログラムは、kcov を使用したテストプログラム内でのカバレッジ収集を示しています：

#include <stdio.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>
#include <sys/types.h>
#include <sys/stat.h>
#include <sys/ioctl.h>
#include <sys/mman.h>
#include <unistd.h>
#include <fcntl.h>

#define KCOV_INIT_TRACE                     _IOR('c', 1, unsigned long)
#define KCOV_ENABLE                 _IO('c', 100)
#define KCOV_DISABLE                        _IO('c', 101)
#define COVER_SIZE                  (64<<10)

#define KCOV_TRACE_PC  0
#define KCOV_TRACE_CMP 1

int main(int argc, char **argv)
{
    int fd;
    unsigned long *cover, n, i;

    /* A single fd descriptor allows coverage collection on a single
     * thread.
     */
    fd = open("/sys/kernel/debug/kcov", O_RDWR);
    if (fd == -1)
            perror("open"), exit(1);
    /* Setup trace mode and trace size. */
    if (ioctl(fd, KCOV_INIT_TRACE, COVER_SIZE))
            perror("ioctl"), exit(1);
    /* Mmap buffer shared between kernel- and user-space. */
    cover = (unsigned long*)mmap(NULL, COVER_SIZE * sizeof(unsigned long),
                                 PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
    if ((void*)cover == MAP_FAILED)
            perror("mmap"), exit(1);
    /* Enable coverage collection on the current thread. */
    if (ioctl(fd, KCOV_ENABLE, KCOV_TRACE_PC))
            perror("ioctl"), exit(1);
    /* Reset coverage from the tail of the ioctl() call. */
    __atomic_store_n(&cover[0], 0, __ATOMIC_RELAXED);
    /* That's the target syscal call. */
    read(-1, NULL, 0);
    /* Read number of PCs collected. */
    n = __atomic_load_n(&cover[0], __ATOMIC_RELAXED);
    for (i = 0; i < n; i++)
            printf("0x%lx\n", cover[i + 1]);
    /* Disable coverage collection for the current thread. After this call
     * coverage can be enabled for a different thread.
     */
    if (ioctl(fd, KCOV_DISABLE, 0))
            perror("ioctl"), exit(1);
    /* Free resources. */
    if (munmap(cover, COVER_SIZE * sizeof(unsigned long)))
            perror("munmap"), exit(1);
    if (close(fd))
            perror("close"), exit(1);
    return 0;
}

複数のスレッドからカバレッジを収集する必要がある場合、各スレッドで別途 /sys/kernel/debug/kcov を開く必要があります。

インターフェースは細かい粒度で設計されており、テストプロセスの効率的なフォークを可能にします。つまり、親プロセスは /sys/kernel/debug/kcov を開き、トレースモードを有効化し、カバレッジバッファを mmap してから、ループで子プロセスをフォークします。子プロセスはカバレッジを有効化するだけで（スレッド終了時に自動的に無効化されます）。

比較演算子の収集¶

比較演算子の収集はカバレッジ収集と同様です：

/* Same includes and defines as above. */

/* Number of 64-bit words per record. */
#define KCOV_WORDS_PER_CMP 4

/*
 * The format for the types of collected comparisons.
 *
 * Bit 0 shows whether one of the arguments is a compile-time constant.
 * Bits 1 & 2 contain log2 of the argument size, up to 8 bytes.
 */

#define KCOV_CMP_CONST          (1 << 0)
#define KCOV_CMP_SIZE(n)        ((n) << 1)
#define KCOV_CMP_MASK           KCOV_CMP_SIZE(3)

int main(int argc, char **argv)
{
    int fd;
    uint64_t *cover, type, arg1, arg2, is_const, size;
    unsigned long n, i;

    fd = open("/sys/kernel/debug/kcov", O_RDWR);
    if (fd == -1)
            perror("open"), exit(1);
    if (ioctl(fd, KCOV_INIT_TRACE, COVER_SIZE))
            perror("ioctl"), exit(1);
    /*
    * Note that the buffer pointer is of type uint64_t*, because all
    * the comparison operands are promoted to uint64_t.
    */
    cover = (uint64_t *)mmap(NULL, COVER_SIZE * sizeof(unsigned long),
                                 PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
    if ((void*)cover == MAP_FAILED)
            perror("mmap"), exit(1);
    /* Note KCOV_TRACE_CMP instead of KCOV_TRACE_PC. */
    if (ioctl(fd, KCOV_ENABLE, KCOV_TRACE_CMP))
            perror("ioctl"), exit(1);
    __atomic_store_n(&cover[0], 0, __ATOMIC_RELAXED);
    read(-1, NULL, 0);
    /* Read number of comparisons collected. */
    n = __atomic_load_n(&cover[0], __ATOMIC_RELAXED);
    for (i = 0; i < n; i++) {
            type = cover[i * KCOV_WORDS_PER_CMP + 1];
            /* arg1 and arg2 - operands of the comparison. */
            arg1 = cover[i * KCOV_WORDS_PER_CMP + 2];
            arg2 = cover[i * KCOV_WORDS_PER_CMP + 3];
            /* ip - caller address. */
            ip = cover[i * KCOV_WORDS_PER_CMP + 4];
            /* size of the operands. */
            size = 1 << ((type & KCOV_CMP_MASK) >> 1);
            /* is_const - true if either operand is a compile-time constant.*/
            is_const = type & KCOV_CMP_CONST;
            printf("ip: 0x%lx type: 0x%lx, arg1: 0x%lx, arg2: 0x%lx, "
                    "size: %lu, %s\n",
                    ip, type, arg1, arg2, size,
            is_const ? "const" : "non-const");
    }
    if (ioctl(fd, KCOV_DISABLE, 0))
            perror("ioctl"), exit(1);
    /* Free resources. */
    if (munmap(cover, COVER_SIZE * sizeof(unsigned long)))
            perror("munmap"), exit(1);
    if (close(fd))
            perror("close"), exit(1);
    return 0;
}

kcov のモード（カバレッジ収集または比較演算子の収集）は相互に排他的です。

リモートカバレッジの収集¶

KCOV_ENABLE では、現在のプロセスから実行されたシステムコールのみのカバレッジが収集されます。KCOV_REMOTE_ENABLE では、kcov_remote_start()/kcov_remote_stop() で注釈付けされたカーネルコードの任意の部分のカバレッジを収集できます。

これにより、2種類のカーネルバックグラウンドスレッドからカバレッジを収集できます：グローバルなスレッド（カーネル起動時に限定されたインスタンス数で生成される、例: USB HCDごとに1つのUSB hub_event()ワーカースレッド）、ローカルなスレッド（ユーザーがカーネルインターフェースとやりとりするときに生成される、例: vhostワーカー）、およびソフト割込みからも収集可能です。

グローバルなバックグラウンドスレッドまたはソフトIRQからカバレッジを収集するためには、一意のグローバルハンドルを割り当て、kcov_remote_start()コールに渡す必要があります。その後、ユーザー空間プロセスは、kcov_remote_arg構造体のhandles配列フィールドに、このようなハンドルのリストをKCOV_REMOTE_ENABLE ioctlの引数として渡すことができます。これにより、使用中のkcovデバイスが、それらのハンドルで参照されるコードセクションにアタッチされます。

複数のローカルバックグラウンドスレッドが異なるユーザー空間プロセスから生成される可能性があるため、注釈ごとに単一のグローバルハンドルを使用することはできません。代わりに、ユーザー空間プロセスは、kcov_remote_arg構造体のcommon_handleフィールドを通じてゼロでないハンドルを渡します。このcommonハンドルは、現在のtask_structのkcov_handleフィールドに保存され、新たに生成されたスレッドにカスタム注釈を通じて渡される必要があります。それらのスレッドは、kcov_remote_start()/kcov_remote_stop()で注釈を付ける必要があります。

内部的には、kcovはハンドルをu64整数として保存します。ハンドルの上位バイトは、そのハンドルが属するサブシステムのIDを示し、下位4バイトはそのサブシステム内のスレッドインスタンスのIDを示します。commonハンドルは特定のサブシステムに属さないため、サブシステムIDとして予約値0が使用されます。バイト4〜7は現在予約されており、ゼロでなければなりません。今後はサブシステムまたはハンドルIDに使用されるバイト数が増える可能性があります。

特定のユーザー空間プロセスがcommonハンドルを通じてカバレッジを収集する場合、kcovは現在のtask_structから取得したkcov_handleを使用して注釈されたコードセクションごとにカバレッジを収集します。ただし、non-commonハンドルは異なるサブシステムから選択的にカバレッジを収集できるため、その違いがあります。

struct kcov_remote_arg {
    __u32           trace_mode;
    __u32           area_size;
    __u32           num_handles;
    __aligned_u64   common_handle;
    __aligned_u64   handles[0];
};

#define KCOV_INIT_TRACE                     _IOR('c', 1, unsigned long)
#define KCOV_DISABLE                        _IO('c', 101)
#define KCOV_REMOTE_ENABLE          _IOW('c', 102, struct kcov_remote_arg)

#define COVER_SIZE  (64 << 10)

#define KCOV_TRACE_PC       0

#define KCOV_SUBSYSTEM_COMMON       (0x00ull << 56)
#define KCOV_SUBSYSTEM_USB  (0x01ull << 56)

#define KCOV_SUBSYSTEM_MASK (0xffull << 56)
#define KCOV_INSTANCE_MASK  (0xffffffffull)

static inline __u64 kcov_remote_handle(__u64 subsys, __u64 inst)
{
    if (subsys & ~KCOV_SUBSYSTEM_MASK || inst & ~KCOV_INSTANCE_MASK)
            return 0;
    return subsys | inst;
}

#define KCOV_COMMON_ID      0x42
#define KCOV_USB_BUS_NUM    1

int main(int argc, char **argv)
{
    int fd;
    unsigned long *cover, n, i;
    struct kcov_remote_arg *arg;

    fd = open("/sys/kernel/debug/kcov", O_RDWR);
    if (fd == -1)
            perror("open"), exit(1);
    if (ioctl(fd, KCOV_INIT_TRACE, COVER_SIZE))
            perror("ioctl"), exit(1);
    cover = (unsigned long*)mmap(NULL, COVER_SIZE * sizeof(unsigned long),
                                 PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
    if ((void*)cover == MAP_FAILED)
            perror("mmap"), exit(1);

    /* Enable coverage collection via common handle and from USB bus #1. */
    arg = calloc(1, sizeof(*arg) + sizeof(uint64_t));
    if (!arg)
            perror("calloc"), exit(1);
    arg->trace_mode = KCOV_TRACE_PC;
    arg->area_size = COVER_SIZE;
    arg->num_handles = 1;
    arg->common_handle = kcov_remote_handle(KCOV_SUBSYSTEM_COMMON,
                                                    KCOV_COMMON_ID);
    arg->handles[0] = kcov_remote_handle(KCOV_SUBSYSTEM_USB,
                                            KCOV_USB_BUS_NUM);
    if (ioctl(fd, KCOV_REMOTE_ENABLE, arg))
            perror("ioctl"), free(arg), exit(1);
    free(arg);

    /*
     * Here the user needs to trigger execution of a kernel code section
     * that is either annotated with the common handle, or to trigger some
     * activity on USB bus #1.
     */
    sleep(2);

    n = __atomic_load_n(&cover[0], __ATOMIC_RELAXED);
    for (i = 0; i < n; i++)
            printf("0x%lx\n", cover[i + 1]);
    if (ioctl(fd, KCOV_DISABLE, 0))
            perror("ioctl"), exit(1);
    if (munmap(cover, COVER_SIZE * sizeof(unsigned long)))
            perror("munmap"), exit(1);
    if (close(fd))
            perror("close"), exit(1);
    return 0;
}
