AArch64 Linuxにおけるメモリタグ拡張（MTE）¶

著者: Vincenzo Frascino <vincenzo.frascino@arm.com>

Catalin Marinas <catalin.marinas@arm.com>

日付: 2020-02-25

このドキュメントでは、AArch64 Linuxにおけるメモリタグ拡張機能（MTE）の実装について説明します。

はじめに¶

ARMv8.5ベースのプロセッサは、メモリタグ拡張機能（MTE）を導入しています。MTEはARMv8.0の仮想アドレスタグ付けTBI（Top Byte Ignore）機能に基づき構築され、ソフトウェアが物理アドレス空間の16バイト単位のグランルごとに4ビットの割当タグにアクセスできるようにします。このようなメモリ範囲はNormal-Taggedメモリ属性でマッピングする必要があります。論理タグは、メモリアクセスに使用される仮想アドレスのビット59〜56から導出されます。MTEが有効なCPUは論理タグと割当タグを比較し、システムレジスタの設定に応じて不一致時に例外を発生させることがあります。

PROT_MTE¶

割当タグにアクセスするには、ユーザープロセスは新しいフラグを使用してアドレス範囲にタグメモリ属性を有効にする必要があります。 prot フラグ mmap() および mprotect():

このようなページがユーザーアドレス空間に最初にマッピングされる際、割当タグは0に設定され、コピーオンライト時に保持されます。 MAP_SHARED 共有がサポートされ、プロセス間で割当タグを共有できます。

初期プロセス状態¶

オン execve()、新しいプロセスは以下の設定を持ちます：

  • PR_TAGGED_ADDR_ENABLE 0に設定（無効）

  • タグチェックモードが選択されていない（タグチェックエラーは無視される）

  • PR_MTE_TAG_MASK 0に設定（すべてのタグが除外される）

  • PSTATE.TCO 0に設定

  • PROT_MTE 初期メモリマップのいずれにも設定されていない

オン fork()、新しいプロセスは親プロセスの設定とメモリマップ属性を継承します（ただし、 madvise() 範囲の MADV_WIPEONFORK はデータとタグがクリアされ（0に設定される）ます。

正しい使用例¶

MTE コード例

/*
 * To be compiled with -march=armv8.5-a+memtag
 */
#include <errno.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/auxv.h>
#include <sys/mman.h>
#include <sys/prctl.h>

/*
 * From arch/arm64/include/uapi/asm/hwcap.h
 */
#define HWCAP2_MTE              (1 << 18)

/*
 * From arch/arm64/include/uapi/asm/mman.h
 */
#define PROT_MTE                 0x20

/*
 * From include/uapi/linux/prctl.h
 */
#define PR_SET_TAGGED_ADDR_CTRL 55
#define PR_GET_TAGGED_ADDR_CTRL 56
# define PR_TAGGED_ADDR_ENABLE  (1UL << 0)
# define PR_MTE_TCF_SHIFT       1
# define PR_MTE_TCF_NONE        (0UL << PR_MTE_TCF_SHIFT)
# define PR_MTE_TCF_SYNC        (1UL << PR_MTE_TCF_SHIFT)
# define PR_MTE_TCF_ASYNC       (2UL << PR_MTE_TCF_SHIFT)
# define PR_MTE_TCF_MASK        (3UL << PR_MTE_TCF_SHIFT)
# define PR_MTE_TAG_SHIFT       3
# define PR_MTE_TAG_MASK        (0xffffUL << PR_MTE_TAG_SHIFT)

/*
 * Insert a random logical tag into the given pointer.
 */
#define insert_random_tag(ptr) ({                       \
        uint64_t __val;                                 \
        asm("irg %0, %1" : "=r" (__val) : "r" (ptr));   \
        __val;                                          \
})

/*
 * Set the allocation tag on the destination address.
 */
#define set_tag(tagged_addr) do {                                      \
        asm volatile("stg %0, [%0]" : : "r" (tagged_addr) : "memory"); \
} while (0)

int main()
{
        unsigned char *a;
        unsigned long page_sz = sysconf(_SC_PAGESIZE);
        unsigned long hwcap2 = getauxval(AT_HWCAP2);

        /* check if MTE is present */
        if (!(hwcap2 & HWCAP2_MTE))
                return EXIT_FAILURE;

        /*
         * Enable the tagged address ABI, synchronous or asynchronous MTE
         * tag check faults (based on per-CPU preference) and allow all
         * non-zero tags in the randomly generated set.
         */
        if (prctl(PR_SET_TAGGED_ADDR_CTRL,
                  PR_TAGGED_ADDR_ENABLE | PR_MTE_TCF_SYNC | PR_MTE_TCF_ASYNC |
                  (0xfffe << PR_MTE_TAG_SHIFT),
                  0, 0, 0)) {
                perror("prctl() failed");
                return EXIT_FAILURE;
        }

        a = mmap(0, page_sz, PROT_READ | PROT_WRITE,
                 MAP_PRIVATE | MAP_ANONYMOUS, -1, 0);
        if (a == MAP_FAILED) {
                perror("mmap() failed");
                return EXIT_FAILURE;
        }

        /*
         * Enable MTE on the above anonymous mmap. The flag could be passed
         * directly to mmap() and skip this step.
         */
        if (mprotect(a, page_sz, PROT_READ | PROT_WRITE | PROT_MTE)) {
                perror("mprotect() failed");
                return EXIT_FAILURE;
        }

        /* access with the default tag (0) */
        a[0] = 1;
        a[1] = 2;

        printf("a[0] = %hhu a[1] = %hhu\n", a[0], a[1]);

        /* set the logical and allocation tags */
        a = (unsigned char *)insert_random_tag(a);
        set_tag(a);

        printf("%p\n", a);

        /* non-zero tag access */
        a[0] = 3;
        printf("a[0] = %hhu a[1] = %hhu\n", a[0], a[1]);

        /*
         * If MTE is enabled correctly the next instruction will generate an
         * exception.
         */
        printf("Expecting SIGSEGV...\n");
        a[16] = 0xdd;

        /* this should not be printed in the PR_MTE_TCF_SYNC mode */
        printf("...haven't got one\n");

        return EXIT_FAILURE;
}
