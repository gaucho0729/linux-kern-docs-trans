PINCTRL (PIN CONTROL)サブシステム¶

このドキュメントでは、Linuxにおけるピン制御サブシステムについて説明します

このサブシステムは以下の処理を行います:

  • 制御可能なピンの列挙と名前付け

  • ピン、パッド、フィンガー（など）のマルチプレクシング、詳しくは下記をご参照ください

  • ピン、パッド、フィンガー（など）の設定、ソフトウェア制御によるバイアス設定やドライブモード、特定のピン（プルアップ/プルダウン、オープンドレイン、ロードキャパシタンスなど）の設定

上位インターフェース¶

PIN CONTROLLER の定義:

  • ピン制御器とは、ピンを制御可能なハードウェア（通常はレジスタの集合）であり、個々のピンまたはピングループに対してマルチプレクシング、バイアス設定、ロードキャパシタンス設定、ドライブ強度設定などを行えることがあります。

PIN の定義:

  • ピンとは、パッド、フィンガー、ボール、または制御したいパッケージの入出力ラインを指し、これらは0..maxpinの符号なし整数で表されます。この番号空間は各ピン制御器ごとにローカルであり、システム内では複数のこのような番号空間が存在する可能性があります。このピン空間は希疏（すくない）な場合もあり、ピンが存在しない番号のギャップがあるかもしれません。

ピン制御器がインスタンス化されると、ピン制御フレームワークにデスクリプタを登録し、このデスクリプタには、このピン制御器が処理するピンを記述するピンデスクリプタの配列が含まれます。

PGA（ピングリッドアレイ）チップの下から見た例:

     A   B   C   D   E   F   G   H

8    o   o   o   o   o   o   o   o

7    o   o   o   o   o   o   o   o

6    o   o   o   o   o   o   o   o

5    o   o   o   o   o   o   o   o

4    o   o   o   o   o   o   o   o

3    o   o   o   o   o   o   o   o

2    o   o   o   o   o   o   o   o

1    o   o   o   o   o   o   o   o

このパッケージ上のすべてのピンを名前を付けて登録するには、ドライバで以下のように実装できます:

#include <linux/pinctrl/pinctrl.h>

const struct pinctrl_pin_desc foo_pins[] = {
        PINCTRL_PIN(0, "A8"),
        PINCTRL_PIN(1, "B8"),
        PINCTRL_PIN(2, "C8"),
        ...
        PINCTRL_PIN(61, "F1"),
        PINCTRL_PIN(62, "G1"),
        PINCTRL_PIN(63, "H1"),
};

static struct pinctrl_desc foo_desc = {
        .name = "foo",
        .pins = foo_pins,
        .npins = ARRAY_SIZE(foo_pins),
        .owner = THIS_MODULE,
};

int __init foo_probe(void)
{
        int error;

        struct pinctrl_dev *pctl;

        error = pinctrl_register_and_init(&foo_desc, <PARENT>,
                                          NULL, &pctl);
        if (error)
                return error;

        return pinctrl_enable(pctl);
}

ピン制御サブシステムおよびPINMUX、PINCONFサブグループ、選択されたドライバを有効にするには、マシンのKconfigエントリからそれらを選択する必要があります。これらは使用されるマシンと非常に密接に関連しているためです。例としてarch/arm/mach-u300/Kconfigをご参照ください。

ピンにはこのような単純な名前ではなく、より複雑な名前が使われます。これはチップのデータシートで確認できます。注意してください、コアのpinctrl.hファイルには、ピンの構造体エントリを生成するための高級マクロであるPINCTRL_PIN()が提供されています。ご覧のように、私は上左隅から0を開始し、下右隅まで63まで番号を振りました。この列挙は任意に選ばれたもので、実際にはドライバ内のレジスタレイアウトなどと一致するように番号体系を慎重に考える必要があります。また、ピン制御器が処理するGPIO範囲とのオフセットの一致も考慮する必要があります。

467パッドのパッドリング（ピンとは異なる）を扱う場合、チップのエッジを回るような列挙方法を使用しました。これは業界標準にも見えるため、（これらのパッドもすべて名前がついていました）

466        105
  .        .
  .        .
358        224
 357 .... 225

ピングループ¶

多くのコントローラーはピンのグループを扱う必要があるため、ピンコントローラーサブシステムはピングループを列挙し、特定のグループに属する実際に列挙されたピンを取得するメカニズムを持っています。

例えば、SPIインターフェースを扱うピングループが{ 0, 8, 16, 24 }、I2Cインターフェースを扱うピングループが{ 24, 25 }にあるとします。

これらの2つのグループは、次のようないくつかの汎用的なpinctrl_opsを実装することでピンコントロールサブシステムに提示されます：

#include <linux/pinctrl/pinctrl.h>

struct foo_group {
        const char *name;
        const unsigned int *pins;
        const unsigned num_pins;
};

static const unsigned int spi0_pins[] = { 0, 8, 16, 24 };
static const unsigned int i2c0_pins[] = { 24, 25 };

static const struct foo_group foo_groups[] = {
        {
                .name = "spi0_grp",
                .pins = spi0_pins,
                .num_pins = ARRAY_SIZE(spi0_pins),
        },
        {
                .name = "i2c0_grp",
                .pins = i2c0_pins,
                .num_pins = ARRAY_SIZE(i2c0_pins),
        },
};


static int foo_get_groups_count(struct pinctrl_dev *pctldev)
{
        return ARRAY_SIZE(foo_groups);
}

static const char *foo_get_group_name(struct pinctrl_dev *pctldev,
                                unsigned selector)
{
        return foo_groups[selector].name;
}

static int foo_get_group_pins(struct pinctrl_dev *pctldev, unsigned selector,
                        const unsigned **pins,
                        unsigned *num_pins)
{
        *pins = (unsigned *) foo_groups[selector].pins;
        *num_pins = foo_groups[selector].num_pins;
        return 0;
}

static struct pinctrl_ops foo_pctrl_ops = {
        .get_groups_count = foo_get_groups_count,
        .get_group_name = foo_get_group_name,
        .get_group_pins = foo_get_group_pins,
};


static struct pinctrl_desc foo_desc = {
...
.pctlops = &foo_pctrl_ops,
};

ピンコントロールサブシステムはまず.get_groups_count()関数を呼び出して、合法的なセレクターの総数を決定し、その後他の関数を呼び出してグループの名前やピンを取得します。グループデータ構造の維持はドライバの責任であり、これは単純な例です。実際にはグループ構造により多くのエントリが必要な場合があります（たとえば各グループに関連する特定のレジスタ範囲など）。

ピン設定¶

ピンは、入力または出力として使用される際の電子的特性に関連して、さまざまな方法でソフトウェアで設定されることがあります。例えば、出力ピンを高インピーダンス（トリステート）に設定して有効に切断することができ、入力ピンをVDDまたはGNDに特定の抵抗値で接続してプルアップまたはプルダウンすることで、ピンが駆動されていない場合や接続されていない場合でも安定した値を持つようにする場合があります。

ピンの設定は、マッピングテーブルに設定エントリを追加することでプログラミング可能です。以下の「ボード/マシン構成」セクションを参照してください。

設定パラメータPLATFORM_X_PULL_UPのフォーマットおよび意味は、ピンコントローラードライバによって完全に定義されます。

ピン設定ドライバは、次のようないくつかのピンコントローラーオプスにおけるピン設定の変更に必要なコールバックを実装します：

#include <linux/pinctrl/pinctrl.h>
#include <linux/pinctrl/pinconf.h>
#include "platform_x_pindefs.h"

static int foo_pin_config_get(struct pinctrl_dev *pctldev,
                unsigned offset,
                unsigned long *config)
{
        struct my_conftype conf;

        ... Find setting for pin @ offset ...

        *config = (unsigned long) conf;
}

static int foo_pin_config_set(struct pinctrl_dev *pctldev,
                unsigned offset,
                unsigned long config)
{
        struct my_conftype *conf = (struct my_conftype *) config;

        switch (conf) {
                case PLATFORM_X_PULL_UP:
                ...
                }
        }
}

static int foo_pin_config_group_get (struct pinctrl_dev *pctldev,
                unsigned selector,
                unsigned long *config)
{
        ...
}

static int foo_pin_config_group_set (struct pinctrl_dev *pctldev,
                unsigned selector,
                unsigned long config)
{
        ...
}

static struct pinconf_ops foo_pconf_ops = {
        .pin_config_get = foo_pin_config_get,
        .pin_config_set = foo_pin_config_set,
        .pin_config_group_get = foo_pin_config_group_get,
        .pin_config_group_set = foo_pin_config_group_set,
};

/* Pin config operations are handled by some pin controller */
static struct pinctrl_desc foo_desc = {
        ...
        .confops = &foo_pconf_ops,
};

GPIOサブシステムとの相互作用¶

GPIOドライバは、同じ物理ピンをピンコントローラーのピンとして登録しているピンに対して、さまざまな種類の操作を実行したい場合があります。

まず第一に、2つのサブシステムは完全に正規直交的に使用できます。詳細については、以下の「ドライバからのピン制御リクエスト」と「ピン制御とGPIOを必要とするドライバ」セクションを参照してください。ただし、一部の状況ではピンとGPIO間のサブシステム横断マッピングが必要です。

ピンコントローラーサブシステムはピンスペースをピンコントローラーごとにローカルに保つため、ピンコントローラーが特定のGPIOピンを制御するかを特定するためのマッピングが必要です。単一のピンコントローラーが複数のGPIO範囲を多重化する場合（典型的には1セットのピンを持つが、内部で複数のGPIOシリコンブロックを備えたSoC）、以下の通りピンコントローラーインスタンスに任意の数のGPIO範囲を追加できます。 struct gpio_chip) 以下のようにピンコントローラーインスタンスに任意の数のGPIO範囲を追加できます:

struct gpio_chip chip_a;
struct gpio_chip chip_b;

static struct pinctrl_gpio_range gpio_range_a = {
        .name = "chip a",
        .id = 0,
        .base = 32,
        .pin_base = 32,
        .npins = 16,
        .gc = &chip_a;
};

static struct pinctrl_gpio_range gpio_range_b = {
        .name = "chip b",
        .id = 0,
        .base = 48,
        .pin_base = 64,
        .npins = 8,
        .gc = &chip_b;
};

{
        struct pinctrl_dev *pctl;
        ...
        pinctrl_add_gpio_range(pctl, &gpio_range_a);
        pinctrl_add_gpio_range(pctl, &gpio_range_b);
}

この複雑なシステムでは1つのピンコントローラーが2つの異なるGPIOチップを制御しています。「チップA」は16ピン、「チップB」は8ピンを持ちます。「チップA」と「チップB」はそれぞれ異なる.pin_baseを有しており、これはGPIO範囲の開始ピン番号を示します。

「チップA」のGPIO範囲はGPIOベース32から開始し、実際のピン範囲も32から開始します。一方、「チップB」はGPIO範囲とピン範囲の開始オフセットが異なります。「チップB」のGPIO範囲はGPIO番号48から開始し、ピン範囲は64から開始します。

GPIO番号を実際のピン番号に変換するにはこの「pin_base」を使用できます。これらは以下のグローバルGPIOピンスペースでマッピングされます:

チップA:

  • GPIO範囲 : [32 .. 47]

  • ピン範囲 : [32 .. 47]

チップB:

  • GPIO範囲 : [48 .. 55]

  • ピン範囲 : [64 .. 71]

上記の例ではGPIOとピンのマッピングが線形であると仮定しています。マッピングがスパースまたはランダムな場合、以下の通り任意のピン番号の配列を範囲にエンコードできます:

static const unsigned range_pins[] = { 14, 1, 22, 17, 10, 8, 6, 2 };

static struct pinctrl_gpio_range gpio_range = {
        .name = "chip",
        .id = 0,
        .base = 32,
        .pins = &range_pins,
        .npins = ARRAY_SIZE(range_pins),
        .gc = &chip;
};

この場合、pin_baseプロパティは無視されます。ピングループの名前がわかっている場合、上記の構造体のpinsとnpins要素はpinctrl_get_group_pins()関数を使用して初期化できます。例えばピングループ「foo」の場合:

pinctrl_get_group_pins(pctl, "foo", &gpio_range.pins,
                       &gpio_range.npins);

ピンコントローラーサブシステム内のGPIO固有の関数が呼び出される際、これらの範囲はピンをすべてのコントローラーのピン範囲と照合して適切なピンコントローラーを検索するために使用されます。一致する範囲を処理するピンコントローラーが見つかった場合、そのピンコントローラーにGPIO固有の関数が呼び出されます。

ピンバイアスやピン多重化などすべての機能において、ピンコントローラーサブシステムは渡されたGPIO番号から対応するピン番号を検索し、範囲の内部情報を用いてピン番号を取得します。その後、サブシステムはそのピン番号をピンコントロールドライバに渡すため、ドライバはその処理範囲内のピン番号を受け取ります。さらに、範囲ID値も渡されるため、ピンコントローラーはどの範囲を処理すべきかを把握できます。

ピンコントロールドライバーからpinctrl_add_gpio_rangeを呼び出すことは非推奨です。pinctrlとGPIOドライバーのバインディング方法については、Documentation/devicetree/bindings/gpio/gpio.txtのセクション2.1を参照してください。

これらの呼び出しはpinmux_*という名前前缀を使用しています。他の呼び出しではこの前缀を使用しないでください。

ピンmuxingとは？¶

PINMUX（パッドmux、ボールmux、代替機能、ミッションモードなど）とは、チップベンダーが電気パッケージを製造する際、アプリケーションに応じて物理ピン（ボール、パッド、フィンガーなど）を複数の排他的な機能に割り当てることを指します。ここでいう「アプリケーション」とは、通常、パッケージを電子システムに実装するためのソルダリングや配線の方法を意味しますが、フレームワークにより実行時にも機能を変更することが可能です。

PGA（ピングリッドアレイ）チップの下から見た例:

     A   B   C   D   E   F   G   H
   +---+
8  | o | o   o   o   o   o   o   o
   |   |
7  | o | o   o   o   o   o   o   o
   |   |
6  | o | o   o   o   o   o   o   o
   +---+---+
5  | o | o | o   o   o   o   o   o
   +---+---+               +---+
4    o   o   o   o   o   o | o | o
                           |   |
3    o   o   o   o   o   o | o | o
                           |   |
2    o   o   o   o   o   o | o | o
   +-------+-------+-------+---+---+
1  | o   o | o   o | o   o | o | o |
   +-------+-------+-------+---+---+

これはテトリスではありません。考えるべきゲームはチェスです。すべてのPGA/BGAパッケージがチェス盤のような配置ではありません。大規模なパッケージでは、設計パターンに応じて「穴」が配置されることがあります。しかし、ここでは簡単な例としてこの比喩を使用しています。見られるピンの中には、いくつかのVCCやGNDでチップに電力を供給するものがあり、多くのピンは外部メモリインターフェースなどの大規模ポートに使用されます。残りのピンはしばしばピンマルチプレクシングの対象になります。

上記の8x8 PGAパッケージでは、ピン番号0から63が物理ピンに割り当てられます。pinctrl_register_pins()を使用して、適切なデータセットに基づきピンを{ A1, A2, A3 … H6, H7, H8 }という名前で指定します。

この8x8 BGAパッケージでは、{ A8, A7, A6, A5 }のピンをSPIポートとして使用できます（これらはCLK、RXD、TXD、FRMの4つのピンです）。その場合、ピンB5は一般的なGPIOピンとして使用できます。しかし、別の設定では{ A5, B5 }のピンをI2Cポートとして使用できます（これらはSCL、SDAの2つのピンです）。言うまでもなく、SPIポートとI2Cポートは同時に使用できません。ただし、パッケージ内部ではSPIロジックを実行するシリコンを、{ G4, G3, G2, G1 }のピンに別のルートで出力することも可能です。

{ A1, B1, C1, D1, E1, F1, G1, H1 }の下段には特別な機能があります。これは2、4、または8ビット幅の外部MMCバスで、それぞれ2、4、または8ピンを消費します。したがって、{ A1, B1 }が使用されるか、{ A1, B1, C1, D1 }が使用されるか、あるいはすべてのピンが使用されるかです。すべての8ビットを使用する場合、{ G4, G3, G2, G1 }のSPIポートは当然ながら使用できません。

このように、チップ内に存在するシリコンブロックは、「muxed」して異なるピン範囲に多重化されます。現代のSoC（システムオンチップ）では、I2C、SPI、SDIO/MMCなどのシリコンブロックが複数含まれており、ピンmux設定によってこれらを異なるピンに接続できます。

一般的な汎用I/Oピン（GPIO）は常に不足しているため、他のI/Oポートに使用されていない場合、ほぼすべてのピンをGPIOピンとして使用できることが一般的です。

ピンmuxingのコンベンション¶

ピンmux機能の目的は、マシン構成でインスタンス化したデバイスにピンmux設定を抽象化して提供することです。clk、GPIO、regulatorサブシステムからインスピレーションを受けているため、デバイスがmux設定を要求しますが、GPIOなど単一のピンを要求することも可能です。

定義:

  • ドライバーは、カーネルのdrivers/pinctrl/*ディレクトリに存在するピン制御サブシステム内に配置され、機能を切り替えることができます。ピン制御ドライバーは可能な機能を把握しており、上記の例ではspi、i2c、mmcの3つのピンmux機能を識別できます。

  • 機能はゼロから一维配列で列挙されると想定されます。この場合、配列は{ spi0, i2c0, mmc0 }のように3つの利用可能な機能を表すことができます。

  • 機能はジェネリックレベルで定義されたピングループと関連付けられます。 always 特定の機能は特定のピングループと関連付けられ、単一のグループでも複数のグループでも可能です。上記の例では、i2c機能は{ A5, B5 }のピンと関連付けられており、コントローラーのピン空間では{ 24, 25 }として列挙されています。

    spi機能はピングループ{ A8, A7, A6, A5 }と{ G4, G3, G2, G1 }に関連付けられており、それぞれ{ 0, 8, 16, 24 }と{ 38, 46, 54, 62 }として列挙されています。

    ピンコントローラーごとにグループ名は一意でなければなりません。同じコントローラー上のグループ名は重複してはなりません。

  • 機能とピングループの組み合わせが特定のピンセットに対する特定の機能を決定します。機能とピングループおよびそれらのマシン固有の詳細情報はピンmuxドライバー内で保持され、外部からはのみ列挙子が知られています。ドライバーコアは以下のものを要求できます。

    • 特定のセレクター（>= 0）を持つ機能の名前

    • 特定の機能に関連するグループの一覧

    • 特定の機能に対してリスト内の特定のグループを有効化する

    すでに上記のように説明したように、ピングループは自己記述的であり、コアはドライバーから特定のグループ内の実際のピン範囲を取得します。

  • 特定のピンコントローラー上の機能（FUNCTIONS）およびグループ（GROUPS）は、ボードファイル、デバイスツリー、または類似のマシン設定構成メカニズムによって特定のデバイスにマッピングされます。これはレギュレータがデバイスに接続される仕組みと同様で、通常は名前によって行われます。機能とグループを定義することによって、特定のデバイスが使用するピンのセットを一意に特定できます。（ある機能に対してのみ1つのピングループが利用可能な場合、グループ名を指定する必要はありません。コアは単に利用可能な最初で唯一のグループを選択します。）

    例として、この特定のマシンがspi0デバイスを使用し、ピンミュクス機能fspi0グループgspi0およびi2c0機能fi2c0グループgi2c0を主なピンコントローラーで使用する場合、マッピングは以下のようになります：

    {
        {"map-spi0", spi0, pinctrl0, fspi0, gspi0},
        {"map-i2c0", i2c0, pinctrl0, fi2c0, gi2c0}
}

    すべてのマッピングにはステート名、ピンコントローラー、デバイス、および機能を割り当てなければなりません。グループは必須ではありません。グループを省略した場合、ドライバーが機能に適用可能な最初のグループが選択されます。これは単純なケースで役立ちます。

    特定のピンコントローラー上の特定の機能に複数のグループをマッピングすることが可能です。これは、特定の機能が異なる設定で異なるピンセットを使用する場合に用いられます。

  • 特定のピンコントローラー上の特定の機能およびグループを使用するピンは、先着順で割り当てられます。他のデバイスのミュクス設定やGPIOピンのリクエストがすでに物理的なピンを占有している場合、そのピンの使用を拒否されます。新しい設定を有効化するには、古い設定を解除（無効化）する必要があります。

ときどき、ドキュメンテーションやハードウェアレジスタはピンではなくパッド（または「指」）を基準にしています。これらはパッケージ内のシリコン上の溶接表面であり、カプセルの下にあるピン/ボールの数と一致しない場合があります。自分にとって意味のある列挙を選びましょう。制御可能なピンのみにエナメラタを定義する場合、それが意味があるならそのようにしてください。

前提条件：

ハードウェアによって可能な機能マッピングのピングループ数が制限されていると仮定します。つまり、電話交換機のように任意の機能が任意のピンにマッピング可能なシステムは存在しないと仮定します。したがって、特定の機能に利用可能なピングループはいくつか（例えば8個程度）に限定され、数百個や任意の選択肢にはなりません。これは入手可能なピンmuxハードウェアを調査した結果確認された特徴であり、ピンmuxドライバーが提示する必要がある前提です。 すべての読者が完了するまで待機したとしても、更新処理が読者がいないと信頼できる期間はゼロである。 サブシステムへの可能機能とピングループのマッピング。

ピンmuxingドライバー¶

ピンmuxコアは、ピンの競合を防ぐことと、ピンコントローラードライバを呼び出してさまざまな設定を実行することを担当します。

ピンmuxドライバの責任は、さらに制限（例えば、負荷による電子的制限などを推論するなど）を課し、要求された機能が実際に許可されるかどうかを判断することです。許可された場合、要求されたMUX設定を実行するため、ハードウェアに変更を指示します。

ピンmuxドライバは、いくつかのコールバック関数を提供する必要があります。一部はオプションです。通常、set_mux()関数が実装され、特定のピンに対して特定のMUX設定を有効にするために、特定のレジスタに値を書き込みます。

ピン制御とGPIOサブシステムの相互作用¶

注意：以下の説明は、Linuxカーネルでgpio_request()や類似関数を使用して特定のピンを使用する使用例を指しています。データシートで「GPIOモード」と呼ばれるものであっても、実際には特定のデバイスの電気的設定に過ぎない場合があります。詳しくは以下の「GPIOモードの陥り」セクションを参照してください。

パブリックピンmux APIには、pinctrl_gpio_request()とpinctrl_gpio_free()という2つの関数があります。これら2つの関数は のみ gpiolibベースのドライバからgpio_request()とgpio_free()のセマンティクスとして呼び出される必要があります。同様に、pinctrl_gpio_direction_[input|output]は、それぞれgpio_direction_[input|output]のgpiolib実装内でのみ呼び出されるべきです。

注意：プラットフォームおよび個々のドライバーは 禁止です GPIOピンを制御するためのリクエストを実行する代わりに、適切なgpiolibドライバを実装し、そのドライバがピンの適切なマトリクス制御およびその他の制御をリクエストするようにする。

個々のピンを他のピンに依存せずGPIOピンに変換できる場合、すべてのピンを関数として定義しようとするアプローチにより、関数リストが非常に長くなる可能性があります。

この場合、GPIO設定およびデバイス関数ごとに関数配列は64項目になることがあります。

そのため、ピン制御ドライバが個々のピンでのみGPIOを有効化できるようにするためには、.gpio_request_enable()および.gpio_disable_free()という2つの関数を実装できます。

この関数は、ピンコントローラーコアが識別する影響を受けるGPIO範囲を渡します。これにより、リクエスト操作で影響を受けるGPIOピンを把握できます。

ドライバがフレームワークからGPIOピンが入力か出力かを示すインジケーションを受けたい場合、.gpio_set_direction()関数を実装できます。この関数はgpiolibドライバから呼び出され、影響を受けるGPIO範囲、ピンオフセット、希望する方向がこの関数に渡されます。

これらの特殊な関数を使用する代わりに、各GPIOピンごとに名前付き関数を使用することも完全に許可されています。pinctrl_gpio_requestは、特殊なGPIOハンドラが登録されていない場合、GPIOピン番号「N」を表す「gpioN」という関数を取得しようと試みます。

GPIOモードの注意点¶

ハードウェアエンジニアが使用する命名規則では、「GPIO」がカーネルが行う意味とは異なるため、開発者がデータシートで「GPIOモード」という表現を見たときに混乱する可能性があります。ハードウェアエンジニアが「GPIOモード」と言う意味は、カーネルインターフェース<linux/gpio.h>で示される使用ケース（カーネルコードからピンを取得し、入力待ちまたは高/低電圧で外部ラインをアサート/クリアする）とは必ずしも一致しないようです。

ハードウェアエンジニアは、「GPIOモード」という言葉を、ピンが他のモード（デバイス用にマトリクス制御されているなど）でない場合に制御できない電気的特性をソフトウェアで制御できる状態を指すと考えています。

ピンとそのピンコントローラー設定およびマトリクス論理との関係におけるGPIO部分は、いくつかの方法で構築できます。以下に2つの例を示します：

(A)
                  pin config
                  logic regs
                  |               +- SPI
Physical pins --- pad --- pinmux -+- I2C
                          |       +- mmc
                          |       +- GPIO
                          pin
                          multiplex
                          logic regs

ピンがGPIOとして使用されているかどうかに関わらず、このピンの電気的特性を設定できます。GPIOをピンに多重化した場合でも、「GPIO」レジスタから高/低レベルを駆動できます。また、ピンは特定のペリフェラルによって制御される一方で、望ましいピン設定プロパティを適用することも可能です。したがって、GPIO機能は他のデバイスがピンを使用する場合でも正規的です。

このような配置では、ピン制御レジスタやGPIOハードウェアモジュールのレジスタは、GPIO駆動専用のメモリ範囲に配置されることが多く、ピン設定やピン多重化を扱うレジスタ範囲は別のメモリ範囲とデータシートの別のセクションに配置されます。

struct pinmux_opsには、「strict」というフラグがあり、このタイプのハードウェアでGPIOとピン多重化消費者が同時に同じピンにアクセスすることをチェックし拒否します。ピン制御ドライバはこのフラグを適切に設定する必要があります。

(B)

                  pin config
                  logic regs
                  |               +- SPI
Physical pins --- pad --- pinmux -+- I2C
                  |       |       +- mmc
                  |       |
                  GPIO    pin
                          multiplex
                          logic regs

このような配置では、GPIO機能は常に有効化できるため、例えばGPIO入力を使ってSPI/I2C/MMC信号を「監視」できるようになります。GPIOブロックで誤った操作を行うと、ピンの通信を妨害する可能性があります。なぜなら、ピンは完全に切断されないからです。GPIO、ピン設定、ピン多重化レジスタが同じメモリ範囲とデータシートの同じセクションに配置される可能性もありますが、必ずしもそうではありません。

一部のピン制御では、物理ピンが(B)と同様に設計されているにもかかわらず、GPIO機能とペリフェラル機能を同時に有効化できない場合があります。このような場合もまた、「strict」フラグを設定し、GPIOと他の多重化されたデバイスの同時起動を拒否する必要があります。

しかし、カーネルの観点ではこれらはハードウェアの異なる側面であり、異なるサブシステムに配置する必要があります：

  • ピンの電気的特性（バイアスやドライブ強度など）を制御するレジスタ（またはレジスタ内のフィールド）は、「ピン設定」設定としてピン制御サブシステムを通じて公開されるべきです。

  • 複数の他のHWブロック（例: I2C、MMC、またはGPIO）からの信号をピンに多重化するレジスタ（またはレジスタ内のフィールド）は、「MUX機能」としてピン制御サブシステムを通じて公開されるべきです。

  • GPIO機能を制御するレジスタ（またはレジスタ内のフィールド）例えばGPIOの出力値を設定する、GPIOの入力値を読み取る、GPIOピンの方向を設定するなどは、GPIOサブシステムを通じて公開されるべきです。また、割り込み機能をサポートする場合、irqchip抽象化を通じて公開されるべきです。

ハードウェアレジスタ設計によっては、GPIOサブシステムが公開する一部の機能はpinctrlサブシステムに呼び出されることがあります。これは、GPIOとピンコントローラーのハードウェアモジュールが分離されているハードウェアで特に必要となる場合があります。例えば、GPIOの方向はピンコントローラーのハードウェアモジュール内のレジスタで決定される場合があります。

バイアスやドライブ強度などのピンの電気的特性は、すべてのケースでピン固有のレジスタに配置されるか、ケース(B)ではGPIOレジスタに含まれることがあります。これはLinuxカーネルが呼ぶ『GPIO』とは必ずしも関係ないとは限りません。

例：ピンは通常UART TXラインとしてマッピングされます。しかしシステムスリープ中にこのピンを『GPIOモード』に設定し、グランドに接続する必要があります。

このピンに対してGPIOサブシステムに1対1でマッピングしようとすると、UART TXとGPIOを同時に使用する必要があると考えるなど複雑な処理が必要になるかもしれません。ピンコントロールハンドルを取得し、UART TXをマッピングするための特定の状態に設定し、その後スリープ中にgpio_direction_output()で低電圧に駆動し、 wakeup時にUART TXに戻すなど、gpio_request/gpio_freeをこのサイクル内で使用するなど、非常に複雑になります。

解決策は、データシートで『GPIOモード』と呼ばれるものに<linux/gpio.h>インターフェースで対応する必要があると考えるのではなく、これを特定のピン設定として捉えることです。例えば<linux/pinctrl/pinconf-generic.h>を確認すると、以下のように記載されています：

PIN_CONFIG_OUTPUT:

この設定によりピンを出力モードに設定し、引数1で高電圧を、引数0で低電圧を指定します。

したがって、通常のピンコントロールマッピング内でピンを『GPIOモード』に設定し、ラインを低電圧に駆動するのも可能です。例えばUARTドライバは以下のようになります：

#include <linux/pinctrl/consumer.h>

struct pinctrl          *pinctrl;
struct pinctrl_state    *pins_default;
struct pinctrl_state    *pins_sleep;

pins_default = pinctrl_lookup_state(uap->pinctrl, PINCTRL_STATE_DEFAULT);
pins_sleep = pinctrl_lookup_state(uap->pinctrl, PINCTRL_STATE_SLEEP);

/* Normal mode */
retval = pinctrl_select_state(pinctrl, pins_default);
/* Sleep mode */
retval = pinctrl_select_state(pinctrl, pins_sleep);

そして、マシン構成は以下のようになります：¶

static unsigned long uart_default_mode[] = {
        PIN_CONF_PACKED(PIN_CONFIG_DRIVE_PUSH_PULL, 0),
};

static unsigned long uart_sleep_mode[] = {
        PIN_CONF_PACKED(PIN_CONFIG_OUTPUT, 0),
};

static struct pinctrl_map pinmap[] __initdata = {
        PIN_MAP_MUX_GROUP("uart", PINCTRL_STATE_DEFAULT, "pinctrl-foo",
                "u0_group", "u0"),
        PIN_MAP_CONFIGS_PIN("uart", PINCTRL_STATE_DEFAULT, "pinctrl-foo",
                        "UART_TX_PIN", uart_default_mode),
        PIN_MAP_MUX_GROUP("uart", PINCTRL_STATE_SLEEP, "pinctrl-foo",
                "u0_group", "gpio-mode"),
        PIN_MAP_CONFIGS_PIN("uart", PINCTRL_STATE_SLEEP, "pinctrl-foo",
                        "UART_TX_PIN", uart_sleep_mode),
};

foo_init(void) {
        pinctrl_register_mappings(pinmap, ARRAY_SIZE(pinmap));
}

ここでは制御したいピンは「u0_group」にあり、「u0」という関数がこのピングループに有効化できるようになっています。その後はUARTの通常の動作に戻ります。ただし、「gpio-mode」という名前の関数も同じピンにマッピング可能で、GPIOモードに移行させることができます。

これによりGPIOサブシステムとの不適切な相互作用は発生せず、望ましい効果を得られます。これはデバイスがスリープ状態に入る際の電気的設定に過ぎず、データシートで「GPIOモード」と呼ばれるものであっても、その点は重要ではありません。UARTデバイスは依然として、そのUARTドライバに関連するピンを制御し、UARTに必要なモードに設定します。Linuxカーネルの意味でのGPIOは単なる1ビットのラインであり、異なる使用ケースです。

「u0」や「gpio-mode」グループのMUX化やプルアップ/プルダウン、出力低電圧設定などのレジスタ操作方法はドライバの責任範囲です。

一部のデータシートでは「GPIOモード」を「低消費電力モード」として記載する場合があり、電気的に同じ意味を持つことが多くあります。しかし後者の場合は、ソフトウェアエンジニアは通常、これは特定のMUX化または設定であるとすぐに識別します。これはGPIO APIと関係ありません。

ボード/マシンの設定¶

ボードとマシンは、完全な稼働システムの構成方法を定義しており、GPIOやデバイスのMUX化、レギュレータの制約、クロックツリーの構造なども含まれます。もちろんピンMUX設定もこの中に含まれます。

マシンのピンコントローラー設定は、シンプルなレギュレータ設定とほぼ同じ形式なので、上記の例配列では第二のファンクションマッピングにI2CとSPIを有効化したいです。

#include <linux/pinctrl/machine.h>

static const struct pinctrl_map mapping[] __initconst = {
        {
                .dev_name = "foo-spi.0",
                .name = PINCTRL_STATE_DEFAULT,
                .type = PIN_MAP_TYPE_MUX_GROUP,
                .ctrl_dev_name = "pinctrl-foo",
                .data.mux.function = "spi0",
        },
        {
                .dev_name = "foo-i2c.0",
                .name = PINCTRL_STATE_DEFAULT,
                .type = PIN_MAP_TYPE_MUX_GROUP,
                .ctrl_dev_name = "pinctrl-foo",
                .data.mux.function = "i2c0",
        },
        {
                .dev_name = "foo-mmc.0",
                .name = PINCTRL_STATE_DEFAULT,
                .type = PIN_MAP_TYPE_MUX_GROUP,
                .ctrl_dev_name = "pinctrl-foo",
                .data.mux.function = "mmc0",
        },
};

ここでのdev_nameは、デバイス構造体を検索するために使用できる一意なデバイス名に一致します（クロックデバイスやレギュレータと同様）。ファンクション名は、このピン範囲を処理するピンMUXドライバが提供する関数と一致する必要があります。

ご覧の通り、システムには複数のピンコントローラーが存在する可能性があり、そのため、マッピングしたい関数を含むピンコントローラーを指定する必要があります。

このピンMUXマッピングをピンMUXサブシステムに登録するには、単に以下のようにします。

ret = pinctrl_register_mappings(mapping, ARRAY_SIZE(mapping));

上記の構造は非常に一般的なので、よりコンパクトな形式にするためのヘルパーマクロがあります。これはpinctrl-fooを使用し、マッピング位置を0と仮定しています。例えば：

static struct pinctrl_map mapping[] __initdata = {
        PIN_MAP_MUX_GROUP("foo-i2c.o", PINCTRL_STATE_DEFAULT,
                          "pinctrl-foo", NULL, "i2c0"),
};

マッピングテーブルには、ピンの設定項目も含まれることがあります。各ピン/グループには複数の設定項目が存在することが一般的であり、テーブルの設定項目は、複数の設定パラメータと値の配列を参照します。以下に、便利なマクロを使用した例を示します：

static unsigned long i2c_grp_configs[] = {
        FOO_PIN_DRIVEN,
        FOO_PIN_PULLUP,
};

static unsigned long i2c_pin_configs[] = {
        FOO_OPEN_COLLECTOR,
        FOO_SLEW_RATE_SLOW,
};

static struct pinctrl_map mapping[] __initdata = {
        PIN_MAP_MUX_GROUP("foo-i2c.0", PINCTRL_STATE_DEFAULT,
                          "pinctrl-foo", "i2c0", "i2c0"),
        PIN_MAP_CONFIGS_GROUP("foo-i2c.0", PINCTRL_STATE_DEFAULT,
                              "pinctrl-foo", "i2c0", i2c_grp_configs),
        PIN_MAP_CONFIGS_PIN("foo-i2c.0", PINCTRL_STATE_DEFAULT,
                            "pinctrl-foo", "i2c0scl", i2c_pin_configs),
        PIN_MAP_CONFIGS_PIN("foo-i2c.0", PINCTRL_STATE_DEFAULT,
                            "pinctrl-foo", "i2c0sda", i2c_pin_configs),
};

さらに、一部のデバイスではマッピングテーブルに特定の名前付き状態が含まれることを期待しています。ハードウェアでピンコントローラーの設定が不要な場合でも、マッピングテーブルにはこれらの名前付き状態を依然として含める必要があります。これにより、状態が提供され、意図的に空であることを明示的に示します。テーブル項目マクロPIN_MAP_DUMMY_STATEは、ピンコントローラーをプログラムすることなく名前付き状態を定義する目的で使用されます：

static struct pinctrl_map mapping[] __initdata = {
        PIN_MAP_DUMMY_STATE("foo-i2c.0", PINCTRL_STATE_DEFAULT),
};

ドライバーからのピン制御リクエスト¶

デバイスドライバーがデバイスをプローブする際、デバイスコアはこれらのデバイスに対して自動的にpinctrl_get_select_default()を実行しようとします。これにより、ドライバー開発者は以下の例のようなボイラープレートコードを追加する必要がありません。ただし、細かい状態選択を行い、「デフォルト」状態を使用しない場合、pinctrlハンドルと状態のデバイスドライバーハンドリングが必要になるかもしれません。

したがって、特定のデバイスのピンをデフォルト状態に設定するだけで済む場合、正しいマッピングテーブルを提供するだけで十分です。デバイスコアが残りの処理を担当します。

個々のドライバーがピン制御を取得・有効化するのを避けるのが一般的です。そのため、可能であれば、プラットフォームコードや、すべての影響を受けるデバイスにアクセスできる場所でピン制御を処理してください。 struct device * ポインタを扱う場合があります。ドライバーが実行時において、例えば異なるMUXマッピング間で切り替える必要がある場合など、このような処理はできません。

典型的なケースは、ドライバーがピンのバイアスを通常動作からスリープ状態に切り替える必要がある場合です。実行時にPINCTRL_STATE_DEFAULTからPINCTRL_STATE_SLEEPへ切り替え、再バイアスや再MUXを行うことで、スリープモードでの電流消費を抑えることができます。

ドライバーは特定の制御状態をアクティブ化するようリクエストできます。通常はデフォルト状態のように以下のようにします：

#include <linux/pinctrl/consumer.h>

struct foo_state {
struct pinctrl *p;
struct pinctrl_state *s;
...
};

foo_probe()
{
        /* Allocate a state holder named "foo" etc */
        struct foo_state *foo = ...;

        foo->p = devm_pinctrl_get(&device);
        if (IS_ERR(foo->p)) {
                /* FIXME: clean up "foo" here */
                return PTR_ERR(foo->p);
        }

        foo->s = pinctrl_lookup_state(foo->p, PINCTRL_STATE_DEFAULT);
        if (IS_ERR(foo->s)) {
                /* FIXME: clean up "foo" here */
                return PTR_ERR(s);
        }

        ret = pinctrl_select_state(foo->s);
        if (ret < 0) {
                /* FIXME: clean up "foo" here */
                return ret;
        }
}

このget/lookup/select/putのシーケンスは、すべてのドライバーが処理する必要がない場合、バスドライバーによって処理することも可能です。バス上の配置がわかっている場合に有効です。

pinctrl APIのセマンティクスは以下の通りです：

  • pinctrl_get()はプロセスコンテキストで呼び出され、クライアントデバイスのすべてのpinctrl情報のハンドルを取得します。これにより、カーネルメモリからstructが割り当てられ、ピンmux状態を保持します。すべてのマッピングテーブルのパースや類似の遅い操作はこのAPI内で行われます。

  • devm_pinctrl_get()はpinctrl_get()の変種であり、取得したポインタに対して関連するデバイスが除去されるときに自動的にpinctrl_put()が呼ばれます。この関数をplain pinctrl_get()よりも推奨します。

  • pinctrl_lookup_state()はプロセスコンテキストで呼び出され、クライアントデバイスの特定の状態のハンドルを取得します。この操作も遅い場合があります。

  • pinctrl_select_state()は、マッピングテーブルで定義された状態に応じてピンコントローラーのハードウェアをプログラミングします。理論上、これはレジスタ設定をハードウェアに直接書き込むだけなので高速なパス操作です。ただし、一部のピンコントローラーのレジスタが遅い/IRQベースのバスにある場合があり、クライアントデバイスは非ブロッキングコンテキストからpinctrl_select_state()を呼び出すことはできません。

  • pinctrl_put()はpinctrlハンドルに関連するすべての情報を解放します。

  • devm_pinctrl_put()はpinctrl_put()の変種であり、devm_pinctrl_get()で返されたpinctrlオブジェクトを明示的に破壊するために使用できます。ただし、この関数の使用はまれで、devm_pinctrl_get()の自動クリーンアップにより、呼び出さなくても処理されます。

    pinctrl_get()はplain pinctrl_put()とペアで使用する必要があります。pinctrl_get()はdevm_pinctrl_put()とペアで使用してはいけません。devm_pinctrl_get()はdevm_pinctrl_put()とオプションでペアで使用できます。devm_pinctrl_get()はplain pinctrl_put()とペアで使用してはいけません。

通常、ピン制御コアはget/putペアを処理し、デバイスドライバの管理操作（利用可能な機能や関連するピンの確認など）を呼び出します。select_stateはピンコントローラードライバに渡され、そのドライバがレジスタを素早く操作してMUX設定を有効化・無効化します。

devm_pinctrl_get()を呼び出すことで、デバイスにピンが割り当てられます。この後、すべてのピンのデバッグfsリストで確認できるはずです。

注意：pinctrlシステムは、要求されたpinctrlハンドルが見つからない場合（例：pinctrlドライバがまだ登録されていない場合）、-EPROBE_DEFERを返します。したがって、ドライバのエラーパスでは、エラーを適切にクリーンアップし、後続の初期化プロセスで再試行できるようにする必要があります。

ピン制御とGPIOを必要とするドライバー¶

また、ドライバが自らピン制御状態を検索・選択することを推奨しませんが、場合によっては避けられないこともあります。

再度、ドライバがリソースを取得する際は次のようになります：

#include <linux/pinctrl/consumer.h>
#include <linux/gpio.h>

struct pinctrl *pinctrl;
int gpio;

pinctrl = devm_pinctrl_get_select_default(&dev);
gpio = devm_gpio_request(&dev, 14, "foo");

ここではまず特定のピン状態を要求し、その後GPIO 14を割り当てます。このようなサブシステムの直交的な使用方法では、GPIOを要求する前に必ずpinctrlハンドルを取得し、望ましいpinctrl状態を選択する必要があります。これは電気的に不快な状況を避けるためのセマンティックな規則です。GPIOサブシステムが処理を開始する前に、ピンをMUXインおよびバイアスする方法を確実に設定したいでしょう。

上記の内容は隠蔽されますが、デバイスコアを使用することで、デバイスがプローブされる直前にピンの設定とMUXがpinctrlコアによって設定されることがあります。これはGPIOサブシステムとは直交しています。

ただし、GPIOサブシステムがpinctrlサブシステムと直接通信する必要があるケースもあります。この場合、pinctrlサブシステムをバックエンドとして使用します。これはGPIOドライバが「ピン制御とGPIOサブシステムの相互作用」セクションで説明した関数を呼び出すときです。これはピンごとのMUXのみを扱い、gpio_*()関数名空間の背後で完全に隠蔽されます。この場合、ドライバはピン制御サブシステムと一切やり取りする必要がありません。

ピンコントロールドライバとGPIOドライバが同じピンを扱い、マルチプレクシングを含む使用ケースがある場合、ピンコントローラーをGPIOドライバのバックエンドとして実装する必要があります。ただし、ハードウェア設計がGPIOコントローラーがピンコントロールシステムと通信することなくハードウェアでピンコントロールのマルチプレクシング状態を上書きできる場合を除きます。

システムピン制御の占有¶

ピンコントロールマップエントリは、ピンコントローラーが登録された際にコアによって占有されることがあります。これは、ピンコントロールデバイスが登録された直後にコアがpinctrl_get()、lookup_state()、select_state()を呼び出すことを意味します。

これは、クライアントデバイス名がピンコントローラーデバイス名と一致し、ステート名がPINCTRL_STATE_DEFAULTの場合のマッピングテーブルエントリに発生します。

{
        .dev_name = "pinctrl-foo",
        .name = PINCTRL_STATE_DEFAULT,
        .type = PIN_MAP_TYPE_MUX_GROUP,
        .ctrl_dev_name = "pinctrl-foo",
        .function = "power_func",
},

主なピンコントローラーで常に適用可能なMUX設定をいくつかコアにリクエストすることが一般的なため、このような用途には便利なマクロが用意されています。

PIN_MAP_MUX_GROUP_HOG_DEFAULT("pinctrl-foo", NULL /* group */,
                              "power_func")

これは上記の構築と同様の結果をもたらします。

ランタイムピンmuxing¶

ランタイムで特定の機能をMUXイン/アウトすることができ、例えばSPIポートをピングループ間で移動する場合があります。例えば上記の例でspi0の場合、同じ機能を異なるピングループで提供し、マッピングの「高度なマッピング」セクションで説明したように名前を変更します。これにより、SPIデバイスには「pos-A」と「pos-B」という2つのステートが存在します。

このスニペットは、foo_probe()で2つのグループのステートオブジェクトを初期化し、その後グループAで定義されたピンにMUXを実行し、最後にグループBで定義されたピンにMUXを実行します。

#include <linux/pinctrl/consumer.h>

struct pinctrl *p;
struct pinctrl_state *s1, *s2;

foo_probe()
{
        /* Setup */
        p = devm_pinctrl_get(&device);
        if (IS_ERR(p))
                ...

        s1 = pinctrl_lookup_state(foo->p, "pos-A");
        if (IS_ERR(s1))
                ...

        s2 = pinctrl_lookup_state(foo->p, "pos-B");
        if (IS_ERR(s2))
                ...
}

foo_switch()
{
        /* Enable on position A */
        ret = pinctrl_select_state(s1);
        if (ret < 0)
        ...

        ...

        /* Enable on position B */
        ret = pinctrl_select_state(s2);
        if (ret < 0)
        ...

        ...
}

上記の処理はプロセスコンテキストから行う必要があります。ピンの確保はステートがアクティブ化されるときに実行されるため、稼働中のシステムでピンが異なる機能に異なるタイミングで使用されることが可能です。
